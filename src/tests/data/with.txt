## with drops
.graph
N Person Alice
N Person Bob
N Person Carol
E KNOWS Alice Bob
E KNOWS Bob Carol
.query
MATCH (a:Person)-[:KNOWS]->(b) WITH a WHERE a.name = 'Alice' RETURN a
.expect
rows 1
a
Alice

## unwind list
.graph
.query
UNWIND ['Ada', 'Bob'] AS n RETURN n
.expect
rows 2
n
Ada
Bob

## second match
.graph
N Person Alice
N Person Bob
N Person Carol
E KNOWS Alice Bob
E KNOWS Bob Carol
.query
MATCH (a:Person {name:'Alice'})-[:KNOWS]->(b) MATCH (b)-[:KNOWS]->(c) RETURN a, b, c
.expect
rows 1
a | b | c
Alice | Bob | Carol

## with distinct
.graph
N Person Alice
N Person Bob
N Person Carol
E KNOWS Alice Bob
E KNOWS Alice Carol
.query
MATCH (a:Person)-[:KNOWS]->(b) WITH DISTINCT a RETURN a
.expect
rows 1
a
Alice

## unwind after match
.graph
N Person Alice
.query
MATCH (a:Person {name:'Alice'}) UNWIND ['x', 'y'] AS t RETURN a, t
.expect
rows 2
a | t
Alice | x
Alice | y
